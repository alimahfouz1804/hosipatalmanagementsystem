cmake_minimum_required(VERSION 3.16)
project(HospitalArraysGUI LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(HospitalOOPArrays main.cpp)
add_library(HospitalStorage hospitalfile.cpp)
add_executable(HospitalArraysGUI WIN32 gui.cpp)
target_link_libraries(HospitalArraysGUI PRIVATE HospitalStorage comctl32 user32 gdi32 shell32)
target_compile_definitions(HospitalArraysGUI PRIVATE _CRT_SECURE_NO_WARNINGS)
set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT HospitalArraysGUI)
enable_testing()
add_executable(HospitalTests validation_tests.cpp)
add_test(NAME HospitalValidation COMMAND HospitalTests)
add_executable(HospitalExtendedTests extended_tests.cpp)
add_test(NAME HospitalExtendedValidation COMMAND HospitalExtendedTests)
add_executable(HospitalFileTests persistence_tests.cpp)
target_link_libraries(HospitalFileTests PRIVATE HospitalStorage)
add_test(NAME HospitalPersistence COMMAND HospitalFileTests)
add_executable(HospitalGuiTests gui_persistence_tests.cpp)
target_link_libraries(HospitalGuiTests PRIVATE HospitalStorage comctl32 user32 gdi32 shell32)
target_compile_definitions(HospitalGuiTests PRIVATE _CRT_SECURE_NO_WARNINGS)
add_test(NAME HospitalGuiPersistence COMMAND HospitalGuiTests)
find_package(Python3 QUIET COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
    add_test(NAME HospitalConsoleMenu
        COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/console_tests.py" "$<TARGET_FILE:HospitalOOPArrays>")
endif()
